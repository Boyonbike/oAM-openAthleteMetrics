     Plan: UserProfile database layer

     Context

     The app tracks athlete performance data. Currently there is no persistent
     user profile (name, biometrics, training zones). This layer stores a
     single-row profile that other features (calorie estimation, HRV zone
     display, weight sync) will read. The weight_kg column is intentionally
     write-protected at the UI level — only updateWeight() touches it, fed from
     the daily weight log.

     ---
     Key codebase facts

     - Package: com.athletedata.openAthleteMetrics
     - Serialization: kotlinx.serialization 1.11.0 (already in project — Gson is
     NOT present)
     - DB version: currently 13; this task bumps it to 14
     - Entity pattern: entity + both mappers (toModel()/toEntity()) live
     together in data/db/XEntity.kt
     - Domain models: plain data classes in data/model/
     - Repository pattern: interface in data/repository/, Room impl named Room*
     - Hilt: @Binds abstract for interfaces, @Provides companion for DAOs

     3. data/db/UserProfileEntity.kt

     Room @Entity, plus toDomain() and UserProfile.toEntity() extension
     functions.
     Serialization of hrZones ↔ hrZonesJson uses Json.encodeToString /
     Json.decodeFromString<List<HrZone>>.
     updatedAt is stamped in the repository, not defaulted here (to avoid clock
     skew surprises on reads).

     @Entity(tableName = "user_profile")
     data class UserProfileEntity(
         @PrimaryKey val id: Int = 1,
         @ColumnInfo(name = "name") val name: String? = null,
         @ColumnInfo(name = "date_of_birth") val dateOfBirth: String? = null,
         @ColumnInfo(name = "biological_sex") val biologicalSex: String? = null,
         @ColumnInfo(name = "height_cm") val heightCm: Double? = null,
         @ColumnInfo(name = "weight_kg") val weightKg: Double? = null,
         @ColumnInfo(name = "stride_length_cm") val strideLengthCm: Double? =
     null,
         @ColumnInfo(name = "wrist_circumference_mm") val wristCircumferenceMm:
     Double? = null,
         @ColumnInfo(name = "resting_metabolic_rate") val restingMetabolicRate:
     Double? = null,
         @ColumnInfo(name = "vo2_max") val vo2Max: Double? = null,
         @ColumnInfo(name = "max_hr") val maxHr: Int? = null,
         @ColumnInfo(name = "hr_zones_json") val hrZonesJson: String? = null,
         @ColumnInfo(name = "updated_at") val updatedAt: Long =
     System.currentTimeMillis(),
     )

     fun UserProfileEntity.toDomain(): UserProfile = UserProfile(
         ...,
         hrZones = hrZonesJson?.let { Json.decodeFromString(it) } ?:
     emptyList(),
     )

     fun UserProfile.toEntity(): UserProfileEntity = UserProfileEntity(
         ...,
         hrZonesJson = hrZones.takeIf { it.isNotEmpty() }?.let {
     Json.encodeToString(it) },
     )

     4. data/db/UserProfileDao.kt

     @Dao
     interface UserProfileDao {
         @Query("SELECT * FROM user_profile WHERE id = 1")
         fun observe(): Flow<UserProfileEntity?>

         @Query("SELECT * FROM user_profile WHERE id = 1")
         suspend fun get(): UserProfileEntity?

         @Insert(onConflict = OnConflictStrategy.REPLACE)
         suspend fun upsert(entity: UserProfileEntity)

         @Query("UPDATE user_profile SET weight_kg = :weightKg, updated_at =
     :updatedAt WHERE id = 1")
         suspend fun updateWeight(weightKg: Double, updatedAt: Long)
     }

     interface UserProfileRepository {

     6. data/repository/RoomUserProfileRepository.kt

     @Singleton, @Inject constructor(private val dao: UserProfileDao).
     - observe() → dao.observe().map { it?.toDomain() }
     - get() → dao.get()?.toDomain()
     - upsert() → stamps updatedAt = System.currentTimeMillis() on the entity
     before saving
     - updateWeight() → dao.updateWeight(weightKg, System.currentTimeMillis())
     - Error logging via Timber.e(...) with rethrow, consistent with
     RoomDailyContextRepository
